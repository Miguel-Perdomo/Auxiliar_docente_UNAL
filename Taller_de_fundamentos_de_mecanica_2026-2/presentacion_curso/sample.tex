	\begin{frame}{Estructura de la Sesión Semanal (2 Horas)}
		\begin{block}{Distribución orientativa del tiempo}
			\begin{itemize}
				\item \textbf{10 min -- Organización:} Registro de asistencia y asignación de problemas del taller previo al tablero.
				\item \textbf{40 min -- Trabajo en Tablero:} Sustentación simultánea por grupos y resolución colaborativa.
				\item \textbf{20 min -- Retroalimentación:} Intervención docente sobre vacíos conceptuales y errores comunes.
				\item \textbf{30 min -- Evaluación / Cierre:} Quiz individual (semanas pares) o introducción práctica al nuevo tema (semanas impares).
				\item \textbf{20 min -- Asignación del Nuevo Taller:} Pautas clave para la guía de la siguiente semana.
			\end{itemize}

	% ----- SLIDE 6: DINÁMICA DE TRABAJO -----
	\begin{frame}{Mecánica Operativa de los Talleres}
		\vspace{-0.2cm}
		\begin{block}{Ciclo Semanal ($N \rightarrow N+1$)}
			\begin{enumerate}
				\item \textbf{Semana $N$:} Se publica la guía de ejercicios (4 a 6 problemas clave).
				\item \textbf{Trabajo Autónomo:} El grupo resuelve el taller de forma colaborativa durante la semana.
				\item \textbf{Semana $N+1$ (En clase):} 
				\begin{itemize}
					\item Se recoge la solución en físico/digital por grupo.
					\item Se realiza sorteo de estudiantes para pasar al tablero a explicar los ejercicios.
					\item Se aplica el quiz correspondiente (en semanas pares).
				\end{itemize}
			\end{enumerate}
		\end{block}
		
		\vspace{0.1cm}
		\begin{alertblock}{Criterio de Calificación de Talleres}
			\small Ningún taller se queda sin calificar: si el grupo no sale al tablero, se evalúa el documento escrito entregado esa misma sesión.
		\end{alertblock}
	\end{frame}

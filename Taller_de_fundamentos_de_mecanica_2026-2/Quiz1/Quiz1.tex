\newcommand{\quizcontent}{%

		\vspace*{0.3cm}
		{\large\bfseries Quiz 1}\\[0.05cm]
		{\normalsize\bfseries Fundamentos de Mecánica (1000019)}
	\end{center}
	
	\vspace{0.1cm}
	
	\begin{enumerate}[leftmargin=*]
		\item Suponga que tiene los vectores $\vec{A} = \hat{i} + 7\hat{j} - 2\hat{k}$, $\vec{B} = 3\hat{i} - 4\hat{j}$ y $\vec{C} = 2\hat{i} + 3\hat{k}$. Determine: $(\vec{A} \times \vec{B}) + \vec{C}$.
		
		\item Un jugador de béisbol, después de ganar un partido, se quita la gorra y la lanza verticalmente hacia arriba con una rapidez inicial de $20$ m/s. La gorra experimenta una desaceleración de $9.8$ m/s$^2$.
		\begin{enumerate}[label=\alph*)]
			\item ¿Cuánto tiempo le toma a la gorra alcanzar su altura máxima?
			\item ¿Cuál es esa altura máxima?
			\item Si la gorra es atrapada a la misma altura desde la que fue lanzada, ¿cuál es su tiempo total de vuelo?
		\end{enumerate}
	\end{enumerate}
}

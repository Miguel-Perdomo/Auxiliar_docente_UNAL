	\section{Leyes de Newton}
	% =========================================================
	
	\noindent Para los siguientes ejercicios halle primero las expresiones algebraicas de las incógnitas y luego utilice los valores dados en cada problema. Use $g = 9.8\ \mathrm{m/s^2}$ y desprecie la fricción y las masas de las cuerdas y de las poleas en todos los problemas.
	
	\vspace{0.1cm}
	
	\begin{enumerate}[label=\arabic*., leftmargin=*, itemsep=8pt]
		
		% ---------- 1 ----------
		\item \begin{minipage}[t]{0.66\linewidth}
			Cuatro masas $m_1 = 6.5$ kg, $m_2 = 3.8$ kg, $m_3 = 10.7$ kg y $m_4 = 4.2$ kg están colgando de un techo, como se muestra en la figura. Estas masas están colgadas mediante cuerdas (sin masa). ¿Cuál es la tensión de la cuerda que conecta las masas $m_1$ y $m_2$?
		\end{minipage}\hfill
		\begin{minipage}[t]{0.28\linewidth}
			\centering
			\begin{tikzpicture}[scale=0.75, baseline=(current bounding box.north)]
				\draw[techo] (-1,0) -- (1,0);
				\fill[pattern=north east lines, pattern color=gray] (-1,0) rectangle (1,0.18);
				\draw[cuerda] (0,0) -- (0,-3.7);
				\foreach \y/\r/\lab in {-0.7/0.16/m_1, -1.5/0.12/m_2, -2.5/0.26/m_3, -3.7/0.17/m_4}{

					\node[anchor=west] at (\r+0.08,\y) {$\lab$};
				}
			\end{tikzpicture}

		% ---------- 2 ----------
		\item La figura muestra una masa colgada de un techo. Halle las tensiones de las cuerdas en las dos configuraciones.
		
		\begin{center}
			% ----- (a) -----
			\begin{tikzpicture}[scale=0.72]
				\def\h{1.5}
				\draw[techo] (-2.2,0) -- (1.8,0);
				\fill[pattern=north east lines, pattern color=gray] (-2.2,0) rectangle (1.8,0.18);
				\coordinate (K) at (0,-\h);
				\coordinate (A) at ({-\h/tan(40)},0);
				\coordinate (B) at ({\h/tan(50)},0);
				\draw[cuerda] (A) -- (K) -- (B);
				\draw[cuerda] (K) -- (0,-3.0);
				\draw[fill=MassBlue] (0,-3.3) circle (0.3);
				\node[anchor=west] at (0.38,-3.3) {$5.00$ kg};
				\fill (K) circle (0.06);
				% ángulos medidos desde el techo
				\draw ($(A)+(0.6,0)$) arc[start angle=0, end angle=-40, radius=0.6];
				\node at ($(A)+({1.0*cos(-20)},{1.0*sin(-20)})$) {\footnotesize $40^\circ$};
				\draw ($(B)+(-0.6,0)$) arc[start angle=180, end angle=230, radius=0.6];
				\node at ($(B)+({1.0*cos(205)},{1.0*sin(205)})$) {\footnotesize $50^\circ$};
				\node[below left] at ($(A)!0.5!(K)$) {$T_1$};
				\node[below right] at ($(B)!0.5!(K)$) {$T_2$};
				\node[left] at (0,-2.25) {$T_3$};
				\node at (0,-4.0) {\small (a)};
			\end{tikzpicture}
			\hspace{1.2cm}
			% ----- (b) -----
			\begin{tikzpicture}[scale=0.72]
				\def\h{1.2}

		% ---------- 3 ----------
		\item Tres bloques de masa $m_1 = 3$ kg, $m_2 = 2$ kg y $m_3 = 5$ kg están interconectados por cuerdas y son halados por una fuerza $\vec{F} = 5\,\hat{\imath}$ N sobre una tabla horizontal sin fricción. Halle la aceleración de las masas y las tensiones de las cuerdas.
		
		\begin{center}
			\begin{tikzpicture}
				\draw[techo] (-0.5,0) -- (6.3,0);
				\fill[pattern=north east lines, pattern color=gray] (-0.5,-0.15) rectangle (6.3,0);
				\foreach \x/\lab in {0/1, 2.2/2, 4.4/3}{
					\draw[fill=BlockGray] (\x,0) rectangle (\x+0.6,0.6);
					\node[above] at (\x+0.3,0.6) {$m_\lab$};
				}
				\draw[cuerda] (0.6,0.3) -- (2.2,0.3);
				\draw[cuerda] (2.8,0.3) -- (4.4,0.3);
				\draw[->, thick] (5.0,0.3) -- (6.1,0.3) node[above] {$\vec{F}$};
			\end{tikzpicture}
		\end{center}
		
		% ---------- 4 ----------
		\item Demuestre que en un plano inclinado (que forma un ángulo $\theta$ con la horizontal) la posición y la rapidez como función del tiempo de una masa que está sobre el plano y que parte de $t=0$, están dadas por
		\begin{equation*}
			x(t) = x_0 + v_{0x}\,t + \dfrac{1}{2}\,g\,t^{2}\sin\theta, \qquad v_x(t) = v_{0x} + g\,t\sin\theta,
		\end{equation*}
		donde el eje $x$ es paralelo al plano y está dirigido hacia abajo, y $x_0$ y $v_{0x}$ son la posición y la velocidad iniciales.
		
		% ---------- 5 ----------
		\item Para los dos bloques de masa $m_1 = 3$ kg y $m_2 = 8$ kg de la figura, calcule las magnitudes de la aceleración de las masas y de la tensión de la cuerda.
		
		\begin{center}
			\begin{tikzpicture}[scale=0.75]
				\draw[techo] (-3.2,0) -- (0,0) -- (0,-2.6);
				\fill[pattern=north east lines, pattern color=gray] (-3.2,-0.18) rectangle (0,0);
				\fill[pattern=north east lines, pattern color=gray] (-0.18,-2.6) rectangle (0,0);
				\draw[fill=PulleyBlue] (0.2,0.15) circle (0.15);
				\draw[fill=BlockGray] (-2.4,0) rectangle (-1.6,0.55);
				\node[above] at (-2.0,0.55) {$m_1$};
				\draw[cuerda] (-1.6,0.3) -- (0.2,0.3) arc[start angle=90, end angle=0, radius=0.15] -- (0.35,-2.1);
				\draw[fill=BlockGray] (0.1,-2.6) rectangle (0.6,-2.1);
				\node[anchor=west] at (0.65,-2.35) {$m_2$};
			\end{tikzpicture}
		\end{center}
		
		% ---------- 6 ----------
		\item En el sistema de la figura, las masas se aceleran con una magnitud de $8\ \mathrm{m/s^2}$. Si $m_1 = 1$ kg y $m_2 = 2$ kg, calcule la magnitud de la tensión de la cuerda y $\sin\beta$.
		
		\begin{center}
			\begin{tikzpicture}[scale=0.7]
				\def\b{28}
				\def\Lb{3.6}
				\coordinate (B) at (0,0);
				\coordinate (C) at (\Lb,0);
				\coordinate (A) at (0,{\Lb*tan(\b)});
				\draw[thick] (B) -- (C) -- (A) -- cycle;
				% pared bajo la polea
				\draw[thick] (C) -- ($(C)+(0,-2.4)$);
				\fill[pattern=north east lines, pattern color=gray] ($(C)+(-0.18,-2.4)$) rectangle (C);
				% bloque m1 sobre el plano
				\coordinate (P) at ($(A)!0.2!(C)$);
				\begin{scope}[shift={(P)}, rotate=-\b]
					\draw[fill=BlockGray] (-0.3,0) rectangle (0.3,0.35);
					\node[rotate=\b] at (0,0.65) {$m_1$};
					\coordinate (S) at (0.3,0.17);
				\end{scope}
				% polea
				\coordinate (Q) at ($(C)+({0.17*sin(\b)},{0.17*cos(\b)})$);
				\draw[cuerda] (S) -- (Q);
				\draw[cuerda] ($(Q)+(0.12,0)$) -- ($(Q)+(0.12,-1.9)$);
				\draw[fill=PulleyBlue] (Q) circle (0.12);
				\draw[fill=BlockGray] ($(Q)+(-0.1,-1.9)$) rectangle ($(Q)+(0.34,-2.35)$);
				\node[anchor=west] at ($(Q)+(0.4,-2.1)$) {$m_2$};
				% ángulo beta
				\draw ($(C)+(-0.8,0)$) arc[start angle=180, end angle={180-\b}, radius=0.8];
				\node at ($(C)+({1.15*cos(180-\b/2)},{1.15*sin(180-\b/2)})$) {$\beta$};
			\end{tikzpicture}
		\end{center}
		
		% ---------- 7 ----------
		\item Dos masas $m_1 = 3$ kg y $m_2 = 2$ kg están unidas por una cuerda como se muestra en la figura. Los bloques se aceleran debido a la diferencia de masas y ángulos. Si los ángulos son $\theta_1 = 30^\circ$ y $\theta_2 = 60^\circ$, ¿cuál es la magnitud de la tensión de la cuerda?
		
		\begin{center}
			\begin{tikzpicture}[scale=0.85]
				\def\thA{30}
				\def\thB{60}
				\def\Lb{5}
				\pgfmathsetmacro{\hh}{\Lb/(1/tan(\thA)+1/tan(\thB))}
				\pgfmathsetmacro{\xa}{\hh/tan(\thA)}
				\coordinate (O) at (0,0);
				\coordinate (C) at (\Lb,0);
				\coordinate (T) at (\xa,\hh);
				\draw[thick] (O) -- (C) -- (T) -- cycle;
				\coordinate (Pl) at ($(T)+(0,0.12)$);
				% bloque m1 (plano izquierdo)
				\coordinate (P1) at ($(O)!0.6!(T)$);
				\begin{scope}[shift={(P1)}, rotate=\thA]
					\draw[fill=BlockGray] (-0.35,0) rectangle (0.35,0.4);
					\node[rotate=-\thA] at (0,0.75) {$m_1$};
					\coordinate (S1) at (0.35,0.2);
				\end{scope}
				% bloque m2 (plano derecho)
				\coordinate (P2) at ($(C)!0.72!(T)$);
				\begin{scope}[shift={(P2)}, rotate=-\thB]
					\draw[fill=BlockGray] (-0.22,0) rectangle (0.22,0.32);
					\node[rotate=\thB] at (0,0.62) {$m_2$};
					\coordinate (S2) at (-0.22,0.16);
				\end{scope}
				\draw[cuerda] (S1) -- (Pl) -- (S2);
				\draw[fill=PulleyBlue] (Pl) circle (0.12);
				% ángulos
				\draw (0.9,0) arc[start angle=0, end angle=\thA, radius=0.9];
				\node at ({1.3*cos(\thA/2)},{1.3*sin(\thA/2)}) {$\theta_1$};
				\draw ($(C)+(-0.6,0)$) arc[start angle=180, end angle={180-\thB}, radius=0.6];
				\node at ($(C)+({0.95*cos(180-\thB/2)},{0.95*sin(180-\thB/2)})$) {$\theta_2$};
			\end{tikzpicture}
		\end{center}
		
		% ---------- 8 ----------
		\item Una esfera de masa $m$ se suspende de un techo con una cuerda de longitud $L$. La esfera da vueltas en un círculo de radio constante $R$, como se muestra en la figura. Demuestre que la rapidez de esta esfera está dada por $v = \sqrt{L\,g\,\sin\alpha\,\tan\alpha}$.
		
		\begin{center}
			\begin{tikzpicture}[scale=0.9]
				\def\al{30}
				\def\Hh{2.5}
				\pgfmathsetmacro{\Rr}{\Hh*tan(\al)}
				% techo
				\draw[techo] (-1.0,0) -- (1.0,0);
				\fill[pattern=north east lines, pattern color=gray] (-1.0,0) rectangle (1.0,0.18);
				\node[anchor=west] at (1.1,0.1) {\small Techo};
				% geometría
				\draw[dotted] (0,0) -- (0,-\Hh);
				\draw[cuerda] (0,0) -- (-\Rr,-\Hh);
				\draw[gray] (0,-\Hh) ellipse ({\Rr} and 0.4);
				\draw[dashed] (0,-\Hh) -- (-\Rr,-\Hh);
				\node[above] at ({-\Rr/2},-\Hh) {\small $R$};
				\draw[->] (-0.6,{-\Hh-0.388}) -- (0.5,{-\Hh-0.388});
				\node[left] at ({-\Rr/2},{-\Hh/2}) {$L$};
				% ángulo alpha
				\draw (0,-0.8) arc[start angle=-90, end angle={-90-\al}, radius=0.8];
				\node at ({1.2*cos(-90-\al/2)},{1.2*sin(-90-\al/2)}) {$\alpha$};
				% esfera y fuerzas
				\draw[->, thick] (-\Rr,-\Hh) -- ++({0.95*sin(\al)},{0.95*cos(\al)}) node[pos=0.55, left=1pt] {$T$};
				\draw[->, thick] (-\Rr,-\Hh) -- ++(0,-0.85) node[right] {$mg$};
				\draw[fill=gray!50] (-\Rr,-\Hh) circle (0.12);
				% diagrama de cuerpo libre
				\begin{scope}[shift={(4.6,-1.9)}]
					\draw[rounded corners=2pt] (-1.9,-1.2) rectangle (2.1,1.7);
					\node[align=center] at (0.1,1.4) {\footnotesize Diagrama de cuerpo libre};
					\coordinate (c) at (-0.2,-0.45);
					\draw[dashed] (c) -- ++(0.58,1.0) node[above right=-2pt] {\footnotesize $T$};
					\draw[->, thick] (c) -- ++(0,1.0) node[left] {\footnotesize $T\cos\alpha$};
					\draw[->, thick] (c) -- ++(0.58,0) node[right] {\footnotesize $T\sin\alpha$};
					\draw[->, thick] (c) -- ++(0,-0.7) node[right] {\footnotesize $mg$};
					\draw[fill=gray!50] (c) circle (0.09);
				\end{scope}
			\end{tikzpicture}
		\end{center}
		
	\end{enumerate}

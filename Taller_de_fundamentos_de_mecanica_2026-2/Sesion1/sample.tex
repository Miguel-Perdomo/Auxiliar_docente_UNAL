% ----- SLIDE 12: VELOCIDAD Y RAPIDEZ MEDIA -----
\begin{frame}{Velocidad y Rapidez Media}
\begin{block}{Velocidad media (cantidad vectorial)}
\[
\vec v_m \equiv \overline{\vec v} = \frac{\Delta \vec x}{\Delta t}
\]
\end{block}
\begin{block}{Rapidez media (cantidad escalar)}
\[
v_m = \frac{d}{\Delta t}
\]
\end{block}
\begin{itemize}
\item Unidades en el S.I.: m/s.
\item Unidades en el sistema inglés: ft/s.
\item Cuando el orden de magnitud es ``grande'': km/h.
\end{itemize}
\end{frame}

% ----- SLIDE 14: SOLUCIÓN NADADOR (1) -----
\begin{frame}{Ejemplo: Nadador Olímpico -- Solución (1)}
Entre $P_1$ ($30$ m, $14$ s) y $P_2$ ($50$ m, $20$ s):
\[
\vec v_m = \frac{\Delta \vec x}{\Delta t} = \frac{(50-30)\text{ m}}{(20-14)\text{ s}}\,\ii = 3.33\ \frac{\text{m}}{\text{s}}\,\ii
\]
\[
v_m = \frac{d}{\Delta t} = 3.33\ \text{m/s}
\]
\end{frame}

% ----- SLIDE 15: SOLUCIÓN NADADOR (2) -----
\begin{frame}{Ejemplo: Nadador Olímpico -- Solución (2)}
Entre el origen ($0$ m, $0$ s) y $P_3$ ($0$ m, $40$ s):
\[
\vec v_m = \frac{\Delta \vec x}{\Delta t} = \frac{(0-0)\text{ m}}{(40-0)\text{ s}}\,\ii = 0\ \frac{\text{m}}{\text{s}}\,\ii
\]
\begin{alertblock}{Nota}
La velocidad media puede ser nula aunque la distancia recorrida $d\neq 0$: el nadador regresó a su punto de partida.
\end{alertblock}
\end{frame}

% ----- SLIDE 18: VELOCIDAD INSTANTÁNEA (definición) -----
\begin{frame}{Velocidad Instantánea}
\[
\vec v(t) = \lim_{\Delta t \to 0} \frac{\Delta \vec x}{\Delta t}\,\ii = \lim_{\Delta t\to 0}\frac{x(t+\Delta t)-x(t)}{\Delta t}\,\ii = \frac{dx}{dt}\,\ii
\]
\end{frame}

% ----- SLIDE 19: EJEMPLO MONEDA -----

\begin{center}

\]
\end{frame}

% ----- SLIDE 21: SUMA DE RIEMANN -----
\begin{frame}{Suma de Riemann e Integrales}
\begin{block}{Integral $\to$ Área bajo la curva}
\[
\int_a^b f(x)\,dx = \lim_{N\to\infty}\sum_{i=1}^N f(x_i)\,\Delta x_i = \lim_{\Delta x_i \to 0}\sum_{i=1}^N f(x_i)\,\Delta x_i
\]
\end{block}
\begin{center}
\begin{tikzpicture}[x=0.85cm,y=0.78cm]

\foreach \a/\b/\h in {0.05/0.75/1, 0.85/1.55/1.8, 1.65/2.35/2.3, 2.45/3.15/2.0, 3.25/3.95/1.2, 4.05/4.75/0.6}{
\filldraw[fill=blue!40, draw=blue!70!black] (\a,0) rectangle (\b,\h);
}

\hspace{0.3cm}

\[

\[
\int_{0}^{t} a_x\,dt' = \int_{v_{0x}}^{v_x(t)} dv_x'
\]
\[
a_x\,t = v_x(t)-v_{0x}

\[
\boxed{v_x(t) = a_x t + v_{0x}}

\begin{frame}{Movimiento con Aceleración Constante (II)}

v_x(t) = a_x t + v_{0x}

\[
\int_{x_0}^{x(t)} dx' = a_x\int_0^t t'\,dt' + v_{0x}\int_0^t dt'

\[
x(t)-x_0 = a_x\frac{t^2}{2}+v_{0x}t

	% =========================================================
	\section{Contexto: el tiempo de reacción}
	% =========================================================
	
	El \textbf{tiempo de reacción simple} es el intervalo que transcurre entre la llegada
	de un estímulo (ver que la regla cae) y el inicio de la respuesta motora (cerrar los
	dedos). En ese intervalo ocurre una cadena de procesos: la señal viaja de la retina al
	cerebro, se procesa, y baja por la médula espinal hasta los músculos del antebrazo.
	
	Es una cantidad de interés real en fisiología, medicina deportiva y seguridad vial.
	Un estudio con 1469 personas entre 18 y 65 años reportó una latencia media de
	\textbf{231 ms} \cite{woods2015}. La prueba de la regla que usaremos aquí es un método
	reconocido para estimar esta cantidad \cite{ferreira2024}.
	
	\subsection*{El principio físico}
	
	Cuando la regla se suelta, cae en caída libre. Si usted la atrapa después de que haya
	caído una distancia $d$, el tiempo transcurrido es:
	
	\begin{equation}
	t = \sqrt{\frac{2d}{g}}, \qquad g = 9.78\ \mathrm{m/s^2} \ \ \text{(Bogotá)}
	\label{eq:caida}
	\end{equation}
	
	\notabox{\textbf{No necesita entender la deducción de esta fórmula todavía} ---
		la verán en el curso magistral. Por ahora úsela como una regla de conversión:
		usted mide una distancia, y la fórmula se la convierte en un tiempo.}

	\begin{enumerate}
		\item Siéntese con el antebrazo apoyado en la mesa y la mano por fuera del borde,
		con los dedos pulgar e índice abiertos unos 3 cm.
		\item Su compañero sostiene la regla verticalmente, de modo que la marca de
		\textbf{0 cm} quede a la altura de sus dedos abiertos.
		\item Sin avisar, su compañero suelta la regla. Usted debe cerrar los dedos
		tan rápido como pueda.
		\item Registre la marca de la regla que queda justo a la altura de sus dedos.
		Esa es la distancia de caída $d$.
		\item Repita el procedimiento \textbf{20 veces}. Registre todos los valores
		en la Tabla \ref{tab:datos}, incluso los que le parezcan ``malos''.
	\end{enumerate}
	
	\notabox{\textbf{Importante:} su compañero debe variar el tiempo de espera entre
		intento e intento (a veces 2 s, a veces 5 s). Si suelta la regla siempre con el
		mismo ritmo, usted aprenderá a anticiparse y estará midiendo otra cosa distinta
		al tiempo de reacción.}

	\section{Análisis}
	% =========================================================
	
	\subsection*{Paso 1: Histograma}
	
	Construya un histograma de sus 20 valores de $d$ en papel milimetrado o cuadriculado.
	Use entre 5 y 7 intervalos de igual ancho. Rotule los ejes con sus unidades.
	
	\vspace{0.2cm}
	\noindent
	\textit{¿La forma se parece a la campana que vimos con los datos de pulso del salón?}
	
	\subsection*{Paso 2: Promedio y desviación estándar}
	
	Calcule, para sus 20 mediciones:
	
	\begin{equation}
	\bar{d} = \frac{1}{N}\sum_{i=1}^{N} d_i
	\qquad\qquad
	\sigma_d = \sqrt{\frac{1}{N}\sum_{i=1}^{N}\left(d_i - \bar{d}\right)^2}
	\end{equation}
	
	\subsection*{Paso 3: Incertidumbre total}
	
	Su medida tiene dos fuentes de incertidumbre independientes:
	
	\begin{itemize}
		\item \textbf{Instrumental:} $\Delta_{\text{inst}} = 0.1$ cm (resolución de la regla).
		\item \textbf{Estadística:} $\sigma_d$, calculada en el paso anterior.
	\end{itemize}
	
	Combínelas en cuadratura:
	
	\begin{equation}
	\Delta d = \sqrt{\left(\sigma_d\right)^2 + \left(\Delta_{\text{inst}}\right)^2}
	\end{equation}
	
	\noindent
	Reporte: $\ d = \bar{d} \pm \Delta d$ \ (en cm, aplicando la regla de cifras significativas).
	
	\subsection*{Paso 4: Conversión a tiempo de reacción}
	
	Convierta su distancia promedio a tiempo usando la ecuación (\ref{eq:caida}).
	\textbf{Trabaje en metros.}
	
	\vspace{0.2cm}
	\noindent
	Para la incertidumbre, use la siguiente expresión (no necesita deducirla):
	
	\begin{equation}
	\Delta t = \frac{\Delta d}{\sqrt{2\,g\,\bar{d}}}
	\end{equation}
	
	\noindent
	Reporte su resultado final en \textbf{milisegundos}: $\ t = \bar{t} \pm \Delta t$ \ [ms].

	\section{Preguntas de análisis}
	% =========================================================
	
	Responda de forma clara y justificada, usando \textbf{sus propios datos}.
	
	\begin{enumerate}
		\item \textbf{¿Cuál incertidumbre domina?} Compare el valor de $\sigma_d$ con el de
		$\Delta_{\text{inst}} = 0.1$ cm. ¿Cuál de las dos contribuye más a $\Delta d$?
		¿Qué significa esto: el problema está en la regla, o en la variabilidad de su
		propia reacción?
		
		\item \textbf{Aleatorio o sistemático.} Suponga que su compañero soltó la regla
		siempre después de contar ``uno, dos, tres''. ¿Qué le pasaría a sus mediciones?
		¿Se volverían más dispersas (error aleatorio) o se desplazarían todas hacia
		valores más pequeños (error sistemático)? Explique.
		
		\item \textbf{Cifras significativas.} Escriba el valor de $\bar{t}$ tal como
		se lo dio la calculadora (con todos sus decimales) y al lado el valor que
		usted reportó finalmente. Explique cuántos decimales eliminó y por qué.
		
		\item \textbf{Comparación con la literatura.} El estudio citado en el contexto
		reporta una media de 231 ms. Compare su resultado con ese valor: ¿quedó por
		encima o por debajo? Proponga al menos una explicación experimental para la
		diferencia (no basta con decir ``me equivoqué'').
		
		\item \textbf{Diseño experimental.} Proponga \textbf{un} cambio concreto en el
		procedimiento que reduciría la incertidumbre de su resultado. Justifique por qué
		funcionaría.
	\end{enumerate}

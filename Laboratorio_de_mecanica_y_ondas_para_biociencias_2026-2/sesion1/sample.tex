% SLIDE 4: ¿QUÉ ES LA INCERTIDUMBRE?
% ============================================================
\begin{frame}{¿Qué es la Incertidumbre?}
	\begin{block}{Toda medida trae un margen de duda}
		\centering
		Viene de dos fuentes distintas:
	\end{block}

	\vspace{0.3cm}
	\begin{itemize}
		\setlength{\itemsep}{10pt}
		\item \textbf{Resolución del instrumento o del conteo}: la separación mínima que se puede distinguir.
		\item \textbf{Repetibilidad}: al repetir la medida, no siempre se obtiene el mismo valor.
	\end{itemize}

	\vspace{0.4cm}
	\begin{exampleblock}{Al final del día, queremos reportar:}
		\centering
		\Large $x = \bar{x} \pm \sigma_x$
	\end{exampleblock}

	\vspace{0.2cm}
	\begin{center}
		\small Vamos paso a paso: primero \textbf{una sola} lectura, después \textbf{muchas} repeticiones.
	\end{center}
\end{frame}

		\centering

	\end{block}

		\textbf{$x = 4.2 \pm 0.1$ cm}
	\end{center}

\begin{frame}{Propagando la Incertidumbre: del Conteo a los BPM}
	\begin{exampleblock}{Misma idea, otro instrumento: contar latidos}
		\centering
		No se pueden contar fracciones de latido $\Rightarrow \Delta N = 1$ latido
		\vspace{0.15cm}

		En 30 s cuentas: $N = 32 \pm 1$ latidos
	\end{exampleblock}

	\vspace{0.3cm}
	\begin{block}{Para pasar a bpm, multiplicamos por una constante}
		\centering
		Si $y = c\cdot x$, entonces $\Delta y = c \cdot \Delta x$
	\end{block}

	\vspace{0.2cm}
	\[
		f = 2 \times (32 \pm 1) = (2\times 32) \pm (2\times 1) = 64 \pm 2 \text{ bpm}
	\]

	\vspace{0.3cm}
	\begin{center}
		\small Este $\pm 2$ bpm es la incertidumbre \textbf{instrumental}: $\Delta f_{\text{inst}} = 2$ bpm. \\
		Guardemos este número.
	\end{center}
\end{frame}

\begin{frame}{Repetimos la Medición: Promedio y Desviación Estándar}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{block}{Fórmulas}
				\[
					\bar{x} = \frac{1}{N}\sum_{i=1}^{N} x_i
				\]
				\[
					\sigma_x = \sqrt{\frac{1}{N}\sum_{i=1}^{N} (x_i - \bar{x})^2}
				\]
			\end{block}
			\small \textcolor{gray}{Un mismo estudiante, 5 conteos de 30 s.}
		\end{column}
		\begin{column}{0.48\textwidth}
			\small
			$N_i = 30,\,35,\,28,\,33,\,29$ \quad ($f_i = 2N_i$)
			\begin{center}
			\begin{tabular}{c|c|c}
				$f_i$ & $f_i-\bar f$ & $(f_i-\bar f)^2$ \\
				\hline
				60 & $-2$ & $4$ \\
				70 & $8$  & $64$ \\
				56 & $-6$ & $36$\\
				66 & $4$  & $16$\\
				58 & $-4$ & $16$\\
				\hline
				& Suma: & $136$
			\end{tabular}
			\end{center}
		\end{column}
	\end{columns}

	\vspace{0.25cm}
	\[
		\bar f = \frac{310}{5} = 62 \text{ bpm} \qquad
		\sigma_{\text{est}} = \sqrt{\frac{136}{5}} = \sqrt{27.2} \approx 5.2 \text{ bpm}
	\]
\end{frame}

% SLIDE 9: COMBINANDO LAS DOS INCERTIDUMBRES (CUADRATURA)
% ============================================================
\begin{frame}{Combinando las Dos Incertidumbres}
	\begin{block}{Tenemos dos contribuciones independientes}
		\centering
		\begin{tabular}{lc}
			Incertidumbre \textbf{instrumental} (resolución del conteo) & $\Delta f_{\text{inst}} = 2$ bpm \\[2pt]
			Incertidumbre \textbf{estadística} (dispersión al repetir) & $\sigma_{\text{est}} \approx 5.2$ bpm \\
		\end{tabular}
	\end{block}

	\vspace{0.3cm}
	\begin{exampleblock}{Se combinan en cuadratura}
		\centering
		\Large $\sigma_{\text{total}} = \sqrt{\sigma_{\text{est}}^{\,2} + \Delta f_{\text{inst}}^{\,2}}$
	\end{exampleblock}

	\vspace{0.2cm}
	\[
		\sigma_{\text{total}} = \sqrt{(5.2)^2 + (2)^2} = \sqrt{27.2 + 4} = \sqrt{31.2} \approx 5.6 \text{ bpm}
	\]

	\vspace{0.2cm}
	\begin{center}
		\small \textbf{¿Por qué en cuadratura y no sumadas?} Porque son errores \textbf{independientes}.

		\vspace{0.1cm}
		\small Note que domina la contribución \textbf{estadística}: $5.2 \rightarrow 5.6$ bpm.
	\end{center}
\end{frame}

% SLIDE 10: REPORTE FINAL — LA REGLA DE ORO
% ============================================================
\begin{frame}{¿Cómo Reporto mi Medida Final?}
	\begin{block}{Regla de oro (en este orden)}
		\begin{enumerate}
			\item Redondear $\sigma_x$ a \textbf{una cifra significativa}.
			\item Redondear $\bar{x}$ a la \textbf{misma posición decimal} que $\sigma_x$.
		\end{enumerate}
		\centering
		\small \textcolor{gray}{El promedio se ajusta a la incertidumbre, nunca al revés.}
	\end{block}

	\begin{exampleblock}{Aplicando la regla a nuestros datos}
		\begin{itemize}
			\item Calculamos: $\bar f = 62.4$ bpm, \ $\sigma_{\text{total}} = 5.6$ bpm
			\item \textbf{Paso 1:} $\sigma_{\text{total}} \approx 6$ bpm \ (una cifra significativa)
			\item \textbf{Paso 2:} $\sigma$ no tiene decimales $\Rightarrow \bar f \approx 62$ bpm
			\item \alert{Reporte final: $f = 62 \pm 6$ bpm}
		\end{itemize}
	\end{exampleblock}

	\begin{center}
		\small \textbf{Error común:} escribir $\mathbf{62.4 \pm 5.6}$ bpm. Si la incertidumbre es de 6 bpm, \\ el ``$.4$'' del promedio es información que no se puede sostener.
	\end{center}
\end{frame}

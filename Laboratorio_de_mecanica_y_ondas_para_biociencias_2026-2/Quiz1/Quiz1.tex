% ----- CONTENIDO DE UN QUIZ -----
\newcommand{\quizcontent}{%
	\begin{center}
		{\large\bfseries\color{UnalGreen} Quiz 1 -- Práctica 1: Plano Inclinado}
	\end{center}
	
	\vspace{0.05cm}
	Nombre: \rule{9cm}{0.4pt} \hfill Grupo: \rule{1.5cm}{0.4pt}
	
	\vspace{0.25cm}
	\begin{enumerate}[leftmargin=*, itemsep=6pt, topsep=2pt]
		\item Sobre un objeto actúa una fuerza neta \emph{constante}.
		¿Cómo se llama el tipo de movimiento que realiza?
		Dibuje su gráfica de velocidad contra tiempo.
		\pts{1.5 pts}
		
		\item
		\begin{minipage}[t]{0.70\textwidth}
			\vspace{0pt}
			Un bloque desliza \emph{sin fricción} por un plano inclinado un
			ángulo $\theta$.
			\begin{enumerate}[label=\alph*), itemsep=3pt, topsep=3pt]
				\item Dibuje el diagrama de cuerpo libre del bloque.
				\pts{1.5 pts}
				\item Su aceleración es $a = g\sin\theta$. ¿Cuánto vale $a$ si
				$\theta = 0^\circ$? ¿Y si $\theta = 90^\circ$?
				\pts{1.0 pt}
				\item Si se cambia el bloque por otro con el doble de masa,
				¿su aceleración aumenta, disminuye o queda igual? Justifique.
				\pts{1.0 pt}
			\end{enumerate}
		\end{minipage}%
		\hfill
		\begin{minipage}[t]{0.25\textwidth}
			\vspace{0pt}
			\centering\plano
		\end{minipage}
	\end{enumerate}
}

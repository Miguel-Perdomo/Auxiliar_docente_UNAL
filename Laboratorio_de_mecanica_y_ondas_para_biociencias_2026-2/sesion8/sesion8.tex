% Barra dentada: 11 orificios (0.5 ... 10.5), longitud 11.
% #1 = x del extremo izquierdo, #2 = y del eje de la barra.
\newcommand{\barra}[2]{%
	\draw[thick, fill=orange!20] (#1,#2-0.25) rectangle (#1+11,#2+0.25);
	\foreach \hi in {0,...,10}{%
		\draw[thin, fill=white] (#1+0.5+\hi,#2) circle (0.09);}%
}
% Masa colgada: #1 = x, #2 = y de la cara inferior de la barra, #3 = etiqueta
\newcommand{\pesa}[3]{%
	\draw[thick] (#1,#2) -- (#1,#2-0.4);
	\draw[thick, fill=StripA!25] (#1-0.4,#2-1.1) rectangle (#1+0.4,#2-0.4);
	\node[font=\scriptsize] at (#1,#2-0.75) {#3};
}
% Apoyo en el orificio de coordenada x = #1 (el orificio queda en y=0)
\newcommand{\fulcro}[1]{%
	\fill[gray!55] (#1,-0.25) -- (#1-0.5,-1.0) -- (#1+0.5,-1.0) -- cycle;
	\draw[thick] (#1-0.8,-1.0) -- (#1+0.8,-1.0);
	\draw[thick, fill=white] (#1,0) circle (0.11);
}
% Dinamometro colgado de un soporte superior (y = #2) que sujeta la barra en x = #1
\newcommand{\dinaarriba}[2]{%
	\draw[line width=2pt, gray!70] (#1-0.7,#2) -- (#1+0.7,#2);
	\draw[thick] (#1,#2) -- (#1,#2-0.2);
	\draw[thick, fill=white] (#1-0.22,#2-1.2) rectangle (#1+0.22,#2-0.2);
	\foreach \dy in {0.4,0.6,0.8,1.0}{\draw[thin] (#1-0.22,#2-\dy) -- (#1-0.08,#2-\dy);}
	\draw[thick] (#1,#2-1.2) -- (#1,0.25);
	\fill (#1,0) circle (0.05);
}
% Dinamometro que tira hacia abajo, anclado en y = #2 (negativo)
\newcommand{\dinaabajo}[2]{%
	\draw[line width=2pt, gray!70] (#1-0.7,#2) -- (#1+0.7,#2);
	\draw[thick] (#1,-0.25) -- (#1,-0.8);
	\draw[thick, fill=white] (#1-0.22,-1.9) rectangle (#1+0.22,-0.8);
	\foreach \dy in {1.1,1.3,1.5,1.7}{\draw[thin] (#1-0.22,-\dy) -- (#1-0.08,-\dy);}
	\draw[thick] (#1,-1.9) -- (#1,#2);
	\fill (#1,0) circle (0.05);
}

	% ============================================================
	% 4. TORQUE
	% ============================================================
	\begin{frame}{El torque: lo que hace girar}
		\begin{columns}[T]
			\begin{column}{0.46\textwidth}
				\vspace{0.3cm}
				Girar depende de \textbf{fuerza por distancia} al punto de giro:
				\[
				\boxed{\;\tau = F\times d\;}
				\]
				\begin{itemize}
					\item $F$: fuerza (N).
					\item $d$: \textbf{brazo}, perpendicular a la línea de la fuerza (m).
					\item $\tau$: torque (N$\cdot$m).
				\end{itemize}
				\vspace{0.1cm}
				Sentido \textbf{horario} o \textbf{antihorario}.
			\end{column}
			\begin{column}{0.54\textwidth}
				\centering
				\begin{tikzpicture}[scale=1.1, every node/.style={font=\normalsize}]

		Torque del dinamómetro $=$ torque de la masa $+$ torque del peso de la barra
		\[
		F_{\text{pred}}\,d_F = P_m\,d_m + P_b\,d_b
		\quad\Longrightarrow\quad
		\boxed{F_{\text{pred}} = \frac{P_m\,d_m + P_b\,d_b}{d_F}}
		\]
		\small
		$P_m = m\,g$ (masa colgada), $P_b = m_b\,g$ (barra). $d_b$: del apoyo al CG; $d_b=0$ si el apoyo está en el CG.

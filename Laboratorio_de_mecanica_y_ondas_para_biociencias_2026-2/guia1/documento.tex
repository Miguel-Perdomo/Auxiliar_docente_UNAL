	\section{Teoría}
	% =========================================================
	
	Un carro que desliza sin fricción por un plano inclinado se mueve con
	aceleración constante $a = g\sin\theta$. En la práctica, el carro no desliza
	sino que \textbf{rueda}, y además hay fricción en los ejes. Ambos efectos hacen
	que la aceleración real sea algo menor:
	
	\begin{equation}
		a = \gamma\,g\,\sin\theta
		\label{eq:principal}
	\end{equation}
	
	\noindent
	donde $\gamma$ es un coeficiente adimensional, menor que 1, que depende de cómo
	está construido el carro (la masa y el radio de sus ruedas). Para un bloque que
	desliza sin fricción, $\gamma = 1$.
	
	\subsection*{Cómo obtener $\theta$ sin transportador}
	
	Medir un ángulo directamente con transportador es impreciso. Es mejor medir dos
	longitudes y usar la definición del seno:
	
	\begin{equation}
		\sin\theta = \frac{h}{L}
		\label{eq:seno}
	\end{equation}
	
	\noindent
	donde $h$ es la altura del extremo elevado del riel sobre la mesa, y $L$ es la
	longitud del riel. Como se trata de un \textbf{cociente}, la incertidumbre se
	propaga con la regla que ya conocen:
	
	\begin{equation}
		\frac{\Delta(\sin\theta)}{\sin\theta}
		= \sqrt{\left(\frac{\Delta h}{h}\right)^{2} + \left(\frac{\Delta L}{L}\right)^{2}}
		\label{eq:propaga}
	\end{equation}
	
	\subsection*{La estrategia de medición}
	
	De la ecuación (\ref{eq:principal}) se ve que si graficamos $a$ contra
	$\sin\theta$ obtenemos una \textbf{recta que pasa por el origen}, de pendiente
	$m = \gamma g$. Por eso vamos a repetir el experimento a varias inclinaciones:
	no para promediar cinco valores sueltos de $g$, sino para obtener uno solo a
	partir de la pendiente de esa recta.

	\section{Procedimiento}
	% =========================================================
	
	\subsection*{6.1 Preparación}
	
	\begin{enumerate}
		\item Midan la longitud $L$ del riel y anoten su incertidumbre $\Delta L$.
		\item Coloquen una regla \textbf{horizontal sobre la mesa}, dentro del campo
		de visión de la cámara. Servirá para calibrar la escala en Tracker.
		\item Ubiquen el montaje en el borde de la mesa y el celular en el costado
		opuesto, de modo que se vea \emph{todo} el recorrido del carro y la regla.
	\end{enumerate}
	
	\subsection*{6.2 Grabación}
	
	\begin{enumerate}
		\item Eleven un extremo del riel a una altura $h$ pequeña. Midan $h$ y
		anoten $\Delta h$.
		\item Suelten el carro desde el reposo, en la parte alta, y graben todo el
		recorrido.
		\item Repitan para \textbf{cinco alturas distintas}, aumentando de a unos
		3 cm. Anoten cada $h$ en su bitácora junto con el nombre del archivo de
		video correspondiente.
	\end{enumerate}
	
	\notabox{Anoten los cinco valores de $h$ \textbf{en el laboratorio}. Si los
		dejan para después, no van a poder reconstruirlos a partir de los videos.}
	
	\subsection*{6.3 Análisis en Tracker}
	
	Para cada video, en Tracker:
	
	\begin{enumerate}
		\item \textbf{Importar el video.} Verifiquen la tasa de cuadros (fps): cada
		cuadro dura $1/\text{fps}$ segundos, y esa es su resolución temporal
		$\Delta t$.
		\item \textbf{Calibrar la escala.} Usen la herramienta \emph{vara de
			calibración} sobre la regla horizontal e introduzcan su longitud real.
		\item \textbf{Fijar el sistema de referencia.} Ubiquen el origen y orienten
		el eje $x$ \textbf{paralelo al riel}.
		\item \textbf{Marcar el carro} cuadro por cuadro (o usar el
		\emph{autotracker}) para generar la tabla de posición contra tiempo.
		\item \textbf{Exportar} la tabla a Excel u hoja de cálculo.
	\end{enumerate}
	
	\notabox{En la sesión analizaremos \textbf{un} video juntos. Los otros cuatro
		quedan como trabajo del grupo. Repártanselos entre los integrantes: marcar
		cuadro por cuadro toma tiempo.}

	\section{Análisis de resultados}
	% =========================================================
	
	\subsection*{7.1 Para cada uno de los cinco videos}
	
	\begin{enumerate}
		\item Grafiquen la posición $x$ contra el tiempo $t$. ¿Qué forma tiene?
		\item Grafiquen la velocidad $v_x$ contra el tiempo $t$. ¿Se ajusta a una
		recta?
		\item Obtengan la aceleración $a$ como la \textbf{pendiente} de la recta
		$v_x$ vs.\ $t$. Reporten también su incertidumbre $\Delta a$.
	\end{enumerate}
	
	\subsection*{7.2 Tabla resumen}

			\hline

			\rule{0pt}{3ex} 4 & & & & & \\ \hline
			\rule{0pt}{3ex} 5 & & & & & \\ \hline
		\end{tabular}
	\end{center}
	
	\noindent
	Usen la ecuación (\ref{eq:propaga}) para llenar la cuarta columna.
	
	\subsection*{7.3 La gráfica principal}
	
	Grafiquen $a$ (eje vertical) contra $\sin\theta$ (eje horizontal), con
	\textbf{barras de error en ambos ejes}. Tracen la recta que mejor represente los
	puntos y obtengan su pendiente $m$.
	
	\vspace{0.15cm}
	\noindent
	Según la ecuación (\ref{eq:principal}), esa pendiente vale $m = \gamma g$.
	
	\subsection*{7.4 Obtengan $\gamma$ y comparen}
	
	Tomando $g = 978$ cm/s$^2$ (valor de la gravedad en Bogotá), calculen
	\begin{equation}
		\gamma = \frac{m}{g}
		\qquad\text{y}\qquad
		\text{discrepancia} = |\gamma - 1|\times 100\%
		\label{eq:gamma}
	\end{equation}
	
	\noindent
	Recuerden que $\gamma = 1$ corresponde a un objeto que desliza sin fricción.

	\begin{frame}{Pregunta 1: energía potencial y cinética}
		\begin{columns}[T]
			\begin{column}{0.48\textwidth}
				\begin{block}{Energía potencial gravitacional}
					\centering
					\Large $U = m\,g\,h$
				\end{block}
				\vspace{0.3cm}
				\begin{block}{Energía cinética}
					\centering
					\Large $K = \tfrac{1}{2}\,m\,v^{2}$
				\end{block}
			\end{column}
			\begin{column}{0.48\textwidth}
				\small
				\textbf{¿Cuándo se conserva la energía mecánica?}
				
				\vspace{0.2cm}
				\begin{exampleblock}{}
					Cuando \textbf{no actúan fuerzas disipativas}: sin fricción, sin
					resistencia del aire, sin choques inelásticos.
				\end{exampleblock}
				
				\vspace{0.3cm}
				En ese caso, toda la energía que pierde un cuerpo en forma de $U$ la
				gana en forma de $K$, y viceversa.
			\end{column}
		\end{columns}
	\end{frame}
	
	% ============================================================
	% SLIDE 5: PREGUNTA 2a
	% ============================================================
	\begin{frame}{Pregunta 2a: rapidez en la cuchilla}
		\small
		La esfera parte del reposo y desciende una altura $h$ hasta la cuchilla.
		No hay fricción apreciable en el hilo, así que la energía se conserva:
		
		\vspace{0.3cm}
		\textbf{Energía en el punto de partida} (reposo, $v=0$):
		\[
		E_i = U = m\,g\,h
		\]
		
		\textbf{Energía justo antes de la cuchilla} (ya bajó toda la altura $h$):
		\[
		E_f = K = \tfrac{1}{2}\,m\,v_0^{2}
		\]
		
		\textbf{Igualando} ($E_i = E_f$):
		\[
		m\,g\,h = \tfrac{1}{2}\,m\,v_0^{2}
		\]
		
		\vspace{0.2cm}
		\begin{exampleblock}{}
			\centering
			\Large $v_0 = \sqrt{2\,g\,h}$
		\end{exampleblock}
	\end{frame}
	
	% ============================================================
	% SLIDE 6: PREGUNTA 2b
	% ============================================================
	\begin{frame}{Pregunta 2b: ¿depende de la masa?}
		\begin{columns}[T]
			\begin{column}{0.55\textwidth}
				\small
				En la ecuación de conservación, la masa aparece en \textbf{ambos
					lados}:
				\[
				\cancel{m}\,g\,h = \tfrac{1}{2}\,\cancel{m}\,v_0^{2}
				\]
				
				\vspace{0.2cm}
				Se cancela, y por eso $v_0 = \sqrt{2gh}$ no contiene $m$.
				
				\begin{exampleblock}{Respuesta}
					\centering
					\textbf{No.} Una esfera pesada y una liviana, soltadas desde la
					misma altura $h$, llegan con la \textbf{misma} rapidez.
				\end{exampleblock}
			\end{column}
			\begin{column}{0.4\textwidth}
				\small
				\textcolor{UnalGreen}{\textbf{Es el mismo resultado}} que en caída
				libre: todos los cuerpos caen igual, sin importar su masa.
				
				\vspace{0.3cm}
				Por eso, en el experimento, \textbf{no es necesario controlar la
					masa} de la esfera con cuidado.
			\end{column}
		\end{columns}
	\end{frame}

	\begin{frame}{Pregunta 2c: cálculo numérico}
		\small
		\textbf{Dato:} $x^{2} = 4\,h\,H$, \quad con $h = 20$ cm y $H = 45$ cm.
		
		\vspace{0.4cm}
		\textbf{Paso 1.} Sustituimos:
		\[
		x^{2} = 4 \times 20 \times 45
		\]
		
		\textbf{Paso 2.} Multiplicamos:
		\[
		x^{2} = 3600 \ \text{cm}^2
		\]
		
		\textbf{Paso 3.} Sacamos raíz cuadrada:
		\[
		x = \sqrt{3600}
		\]
		
		\vspace{0.2cm}
		\begin{exampleblock}{}
			\centering
			\Large $x = 60$ cm
		\end{exampleblock}
	\end{frame}

	% SLIDE 10: CÓMO MEDIR x CORRECTAMENTE
	% ============================================================
	\begin{frame}{Un detalle importante: ¿dónde se mide $x$?}
		\begin{columns}[T]
			\begin{column}{0.48\textwidth}
				\small
				El papel carbón deja una \textbf{mancha}, no un punto. La esfera
				toca, rueda un poco y se detiene.
				
				\vspace{0.3cm}
				\begin{alertblock}{Regla}
					\centering
					Midan al \textbf{centro} de la mancha, no a un borde.
				\end{alertblock}
				
				\vspace{0.3cm}
				La ecuación $x^2=4hH$ describe la trayectoria del \textbf{centro}
				de la esfera, así que hay que estimar dónde cayó ese centro.
			\end{column}
			\begin{column}{0.48\textwidth}
				\small
				\textbf{La incertidumbre de $x$} no es solo la resolución de la
				regla: es también el \textbf{ancho de la mancha}.
				
				\vspace{0.2cm}
				\[
				\Delta x \approx \tfrac{1}{2}\,(\text{ancho de la mancha})
				\]
				
				\vspace{0.3cm}
				\textbf{El otro extremo:} $x$ se mide desde el punto del piso
				justo debajo del \textbf{centro} de la esfera en el momento del
				corte.
			\end{column}
		\end{columns}
	\end{frame}
	
	% ============================================================
	% SLIDE 11: LA LINEALIZACIÓN
	% ============================================================
	\begin{frame}{Por qué graficamos $X$ y $Y$, y no $h$ y $x$}
		\begin{columns}[T]
			\begin{column}{0.46\textwidth}
				\small
				Combinando $v_0=\sqrt{2gh}$ con la caída de la esfera desde $H$
				se llega a
				\[
				x^{2} = 4\,h\,H
				\]
				
				\vspace{0.2cm}
				Definiendo
				\[
				X = \frac{h}{H}
				\qquad
				Y = \left(\frac{\bar{x}}{H}\right)^{2}
				\]
				
				\vspace{0.1cm}
				se obtiene una \textbf{recta por el origen}:
				\[
				Y = 4\,X
				\]
			\end{column}
			\begin{column}{0.5\textwidth}
				\begin{center}
					\begin{tikzpicture}
						\begin{axis}[width=6.2cm, height=4.8cm,
							xlabel={$X = h/H$}, ylabel={$Y = (\bar{x}/H)^2$},
							xmin=0, xmax=0.55, ymin=0, ymax=2.2,
							xtick={0,0.1,...,0.5}, ytick={0,0.5,1,1.5,2},
							grid=both, grid style={gray!30, thin},
							legend style={font=\tiny, at={(0.03,0.97)}, anchor=north west,
								draw=none, fill=white, fill opacity=0.8, text opacity=1}]
							\addplot[domain=0:0.55, samples=2, thick, dashed, nyugray] {4*x};
							\addlegendentry{teórica ($p=4$)}
							\addplot[only marks, mark=*, mark size=1.8pt, red!75!black,
							error bars/.cd, y dir=both, y explicit]
							coordinates {
								(0.10,0.38) +- (0,0.06)
								(0.20,0.74) +- (0,0.06)
								(0.30,1.12) +- (0,0.07)
								(0.40,1.49) +- (0,0.08)
								(0.50,1.86) +- (0,0.09)
							};
							\addlegendentry{datos}
						\end{axis}
					\end{tikzpicture}
				\end{center}
				\vspace{-0.2cm}
				\begin{center}\scriptsize (datos ilustrativos)\end{center}
			\end{column}
		\end{columns}

	% SLIDE 12: PROCEDIMIENTO
	% ============================================================
	\begin{frame}{Procedimiento}
		\begin{enumerate}
			\setlength{\itemsep}{6pt}
			\small
			\item Ubiquen la cuchilla horizontal, justo arriba de la esfera en el
			punto más bajo.
			\item Midan $H$ y \textbf{manténganla constante} todo el experimento.
			\item Marquen en el piso el punto justo debajo de la cuchilla (con
			plomada): desde ahí se mide $x$.
			\item Hagan un lanzamiento de prueba y coloquen ahí el papel carbón.
			\item Suelten la esfera \textbf{desde el reposo}, sin empujarla, desde
			una altura $h_1$.
			\item Midan $h = h_1 - h_2$ y la distancia $x$ (al centro de la
			mancha).
			\item Repitan \textbf{3 veces} para la misma altura.
			\item Repitan todo para \textbf{6 alturas} diferentes.
		\end{enumerate}
		
		\vspace{0.2cm}
		\begin{alertblock}{}
			\centering
			Un impulso al soltar la esfera agrega energía cinética que la
			fórmula no contempla: \textbf{suéltenla, no la empujen}.
		\end{alertblock}
	\end{frame}

	\begin{frame}{Pregunta 1: fuerza constante}
		\begin{columns}[T]
			\begin{column}{0.5\textwidth}
				\small
				\textbf{Pregunta:} si la fuerza neta sobre un objeto es constante,
				¿qué movimiento realiza?
				
				\vspace{0.3cm}
				Por la segunda ley de Newton:
				\[
				\vec{F} = m\,\vec{a}
				\quad\Longrightarrow\quad
				\vec{a} = \frac{\vec{F}}{m} = \text{constante}
				\]
				
				\begin{exampleblock}{Respuesta}
					\centering
					Movimiento \textbf{uniformemente acelerado} (MUA).
				\end{exampleblock}
				
				\vspace{0.2cm}
				Como $v(t) = v_0 + a\,t$, la gráfica $v$ vs.\ $t$ es una
				\textbf{recta} cuya pendiente es la aceleración.
			\end{column}
			\begin{column}{0.48\textwidth}
				\begin{center}
					\begin{tikzpicture}
						\begin{axis}[width=6.2cm, height=5.0cm,
							xlabel={$t$ (s)}, ylabel={$v$ (m/s)},
							xmin=0, xmax=5, ymin=0, ymax=6,
							xtick={0,1,2,3,4,5}, ytick={0,2,4,6},
							grid=both, grid style={gray!30, thin}]
							\addplot[domain=0:5, samples=2, very thick, UnalGreen] {0.5 + 1.0*x};
							\draw[dashed, red!75!black, thick] (axis cs:1.5,2.0) -- (axis cs:3.5,2.0)
							-- (axis cs:3.5,4.0);
							\node[below, font=\scriptsize, red!75!black] at (axis cs:2.5,2.0) {$\Delta t$};
							\node[right, font=\scriptsize, red!75!black] at (axis cs:3.5,3.0) {$\Delta v$};
							\node[font=\scriptsize, fill=white, inner sep=2pt] at (axis cs:1.6,4.7)
							{$a = \dfrac{\Delta v}{\Delta t}$};
						\end{axis}
					\end{tikzpicture}
				\end{center}
			\end{column}
		\end{columns}
	\end{frame}

	% SLIDE 6: DE DONDE SALE a = g sin(theta)
	% ============================================================
	\begin{frame}{¿De dónde sale $a = g\sin\theta$?}
		\begin{columns}[T]
			\begin{column}{0.48\textwidth}
				\begin{block}{Eje $y$ (perpendicular al plano)}
					No hay movimiento en esa dirección:
					\[
					N - mg\cos\theta = 0
					\]
					\[
					\Rightarrow\quad N = mg\cos\theta
					\]
				\end{block}
			\end{column}
			\begin{column}{0.48\textwidth}
				\begin{block}{Eje $x$ (a lo largo del plano)}
					La única fuerza es la componente del peso:
					\[
					mg\sin\theta = m\,a
					\]
					\[
					\Rightarrow\quad a = g\sin\theta
					\]
				\end{block}
			\end{column}
		\end{columns}
		
		\vspace{0.5cm}
		\begin{exampleblock}{}
			\centering
			\Large $a = g\sin\theta$
		\end{exampleblock}
	\end{frame}

	\begin{frame}{Preguntas 2b y 2c: analizando la fórmula}
		\begin{columns}[T]
			\begin{column}{0.48\textwidth}
				\textbf{2b. Casos extremos}
				
				\vspace{0.2cm}
				\small
				$\theta = 0^\circ$: \quad $a = g\sin 0^\circ = 0$
				
				\hspace{1.2cm}Plano horizontal: el bloque no se mueve.
				
				\vspace{0.3cm}
				$\theta = 90^\circ$: \quad $a = g\sin 90^\circ = g$
				
				\hspace{1.3cm}Plano vertical: \textbf{caída libre}.
				
				\vspace{0.4cm}
				\begin{center}
					\textcolor{UnalGreen}{\textbf{La fórmula recupera los dos casos que ya conocemos.}}
				\end{center}
			\end{column}
			\begin{column}{0.48\textwidth}
				\textbf{2c. ¿Depende de la masa?}
				
				\vspace{0.2cm}
				\small
				En la ecuación de Newton, la masa aparece a \textbf{ambos lados}:
				\[
				\cancel{m}g\sin\theta = \cancel{m}a
				\]
				
				\begin{exampleblock}{Respuesta}
					\centering
					Con el doble de masa, la aceleración \textbf{queda igual}.
				\end{exampleblock}
				
				\vspace{0.2cm}
				Un bloque pesado y uno liviano bajan \emph{juntos}, igual que en
				caída libre.
			\end{column}
		\end{columns}
	\end{frame}

	% SLIDE 9: EL CARRO NO ES UN BLOQUE
	% ============================================================
	\begin{frame}{En el laboratorio: el carro no es un bloque}
		\begin{columns}[T]
			\begin{column}{0.5\textwidth}
				\small
				El carro \textbf{rueda} en vez de deslizar, y sus ejes tienen algo de
				fricción. Parte de la energía se va en hacer girar las ruedas.
				
				\vspace{0.3cm}
				Por eso la aceleración real es un poco menor:
				\begin{exampleblock}{}
					\centering
					\Large $a = \gamma\,g\sin\theta$
				\end{exampleblock}
				
				\vspace{0.2cm}
				\begin{itemize}
					\item $\gamma = 1$: bloque ideal sin fricción.
					\item $\gamma < 1$: carro real.
				\end{itemize}
			\end{column}
			\begin{column}{0.46\textwidth}
				\begin{alertblock}{Objetivo de hoy}
					\centering
					Medir $\gamma$ para el carro del laboratorio.
				\end{alertblock}
				
				\vspace{0.4cm}
				\small
				Para eso necesitamos dos cosas:
				\begin{enumerate}
					\item La aceleración $a$ del carro \\
					\textcolor{nyugray}{$\rightarrow$ con Tracker}
					\item El ángulo $\theta$ del riel \\
					\textcolor{nyugray}{$\rightarrow$ con una regla}
				\end{enumerate}
			\end{column}
		\end{columns}
	\end{frame}

						\draw[|-|, thick] (6.6,0) -- (6.6,1.8) node[midway, right] {$h$};
						\draw[|-|, thick] (0.15,0.55) -- (5.85,2.26);
						\node[above left] at (3.0,1.35) {$L$};
						\draw (1.3,0) arc (0:16.7:1.3);
						\node at (1.6,0.22) {$\theta$};
					\end{tikzpicture}
				\end{center}
				
				\vspace{0.2cm}
				\begin{exampleblock}{}
					\centering
					\Large $\sin\theta = \dfrac{h}{L}$
				\end{exampleblock}
			\end{column}
			\begin{column}{0.46\textwidth}
				\small
				Un transportador tiene resolución de $1^\circ$: para ángulos pequeños
				eso es un error enorme.
				
				\vspace{0.3cm}
				Medir dos \textbf{longitudes} con una regla es mucho más preciso.
				
				\vspace{0.3cm}
				Como es un \textbf{cociente}, la incertidumbre se propaga con la regla
				del Taller 2:
				\[
				\frac{\Delta(\sin\theta)}{\sin\theta}
				= \sqrt{\left(\frac{\Delta h}{h}\right)^{2}
					+ \left(\frac{\Delta L}{L}\right)^{2}}
				\]
			\end{column}
		\end{columns}
	\end{frame}
